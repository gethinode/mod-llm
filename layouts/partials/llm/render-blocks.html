{{- $page := . -}} {{- $out := "" -}} {{- range .Params.content_blocks -}} {{- $name := ._bookshop_name -}} {{- $i18nKey := printf "llm-block-%s" $name -}} {{- $title := or (i18n $i18nKey) ($name | title) -}} {{- with .heading }}{{- with .title }}{{- $title = . -}}{{- end }}{{- end -}} {{- with .heading }}{{- with .preheading }}{{- $title = printf "%s — %s" . $title -}}{{- end }}{{- end -}} {{- $body := "" -}} {{- with .heading }}{{- with .content -}} {{- $processed := partial "llm/safe-content.html" (dict "Page" $page "Content" .) -}} {{- $body = printf "%s%s\n\n" $body $processed -}} {{- end }}{{- end -}} {{- $partial := printf "bookshop/components/%s/%s.hugo.md" $name $name -}} {{- if templates.Exists (printf "partials/%s" $partial) -}} {{- $processed := partial "llm/safe-content.html" (dict "Page" $page "Content" (printf "%s" (partial $partial .))) -}} {{- $body = printf "%s%s\n\n" $body $processed -}} {{- end -}} {{- /* A cards or panels block without a heading is only a layout of its items, and its fallback title names the layout. Print no section and raise the items one level. */ -}} {{- $authored := false -}} {{- with .heading }}{{- $authored = or .title .preheading .content -}}{{- end -}} {{- $headless := and (in (slice "cards" "panels") $name) (not $authored) -}} {{- if and $headless (findRE `(?m)^### ` $body 1) -}} {{- $out = printf "%s\n%s" $out ($body | replaceRE `(?m)^#(#{2,5} )` "$1") -}} {{- else if strings.TrimSpace $body -}} {{- $out = printf "%s\n## %s\n\n%s" $out $title $body -}} {{- end -}} {{- end -}} {{- $out = $out | replaceRE `\n{3,}` "\n\n" | strings.TrimLeft "\n" -}} {{ $out | safeHTML }}
